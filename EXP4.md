CREATE TABLE FLIGHT (
    FLIGHT_ID        NUMBER(5) NOT NULL,
    FLIGHT_NO        VARCHAR2(15) NOT NULL,
    AIRLINE          VARCHAR2(50) NOT NULL,
    ORIGIN_CODE      VARCHAR2(10) NOT NULL,
    DESTINATION_CODE VARCHAR2(10) NOT NULL,
    SCHEDULED_TIME   DATE NOT NULL,
    STATUS           VARCHAR2(20) NOT NULL
);
