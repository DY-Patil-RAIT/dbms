CREATE OR REPLACE TRIGGER FLIGHT_STATUS_TRG
BEFORE UPDATE OF STATUS ON FLIGHT
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE(
        'Flight ' || :OLD.FLIGHT_NO ||
        ' status changed from ' || :OLD.STATUS ||
        ' to ' || :NEW.STATUS
    );
END;
/

SELECT TRIGGER_NAME,
       STATUS
FROM USER_TRIGGERS
WHERE TRIGGER_NAME = 'FLIGHT_STATUS_TRG';

UPDATE FLIGHT
SET STATUS = 'BOARDING'

CREATE OR REPLACE TRIGGER BOOKING_INSERT_TRG
AFTER INSERT ON BOOKING
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE(
        'Booking created: ' || :NEW.PNR ||
        ' for Flight ID: ' || :NEW.FLIGHT_ID
    );
END;
/
